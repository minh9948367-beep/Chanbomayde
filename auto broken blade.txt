-- =========================================================
-- HÀM LẤY TÊN FILE TỪ TEXTBOX
-- =========================================================
local function getTargetFileName(text)
    local cleaned = text:gsub("%s+", "") -- Xóa khoảng trắng thừa
    if cleaned == "" then
        return "default_cps.txt" -- File mặc định nếu để trống
    end
    return cleaned .. ".txt"
end

-- =========================================================
-- HÀM FILE SYSTEM (TẢI, GHI, XÓA THEO TÊN FILE)
-- =========================================================
local function saveToDevice(fileName)
    if writefile then
        local rawTable = {}
        for _, pos in ipairs(checkpoints) do
            table.insert(rawTable, {pos.X, pos.Y, pos.Z})
        end
        writefile(fileName, HttpService:JSONEncode(rawTable))
    end
end

local function loadFromDevice(fileName)
    if isfile and readfile and isfile(fileName) then
        local success, decoded = pcall(function()
            return HttpService:JSONDecode(readfile(fileName))
        end)
        if success and type(decoded) == "table" then
            checkpoints = {}
            for _, posArr in ipairs(decoded) do
                table.insert(checkpoints, Vector3.new(posArr[1], posArr[2], posArr[3]))
            end
            return true
        end
    end
    return false
end

-- 2. TÍNH NĂNG KÉO THẢ TRÊN DI ĐỘNG
-- =========================================================
local dragging, dragInput, dragStart, startPos
local function update(input)
    local delta = input.Position - dragStart
    mainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
end
mainFrame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = mainFrame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then dragging = false end
        end)
    end
end)
mainFrame.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then dragInput = input end
end)
UIS.InputChanged:Connect(function(input)
    if input == dragInput and dragging then update(input) end
end)

-- 7. LOGIC TELE VÀ ĐI TUẦN THÔNG MINH
-- =========================================================
local function tryInteract(obj)
    local prompt = obj:FindFirstChildWhichIsA("ProximityPrompt", true)
    if prompt and prompt.Enabled then
        prompt.HoldDuration = 0 
        if fireproximityprompt then
            fireproximityprompt(prompt)
        else
            prompt:InputHoldBegin()
            task.wait(0.05)
            prompt:InputHoldEnd()
        end
    end
end

task.spawn(function()
    while true do
        task.wait(0.1)
        
        if running then
            local character = player.Character
            local hrp = character and character:FindFirstChild("HumanoidRootPart")
            
            if hrp then
                local foundTarget = false
                
                for _, obj in ipairs(workspace:GetDescendants()) do
                    if not running then break end
                    
                    if obj.Name == targetName then
                        local prompt = obj:FindFirstChildWhichIsA("ProximityPrompt", true)
                        
                        if prompt and prompt.Enabled then
                            foundTarget = true
                            
                            local targetCFrame = nil
                            if obj:IsA("Model") then
                                targetCFrame = obj:GetPivot()
                            elseif obj:IsA("BasePart") then
                                targetCFrame = obj.CFrame
                            end
                            
                            if targetCFrame then
                                hrp.CFrame = targetCFrame * CFrame.new(0, 3, 0)
                                task.wait(TELE_DELAY_ITEM) 
                                
                                tryInteract(obj)
                                task.wait(0.2) 
                                break 
                            end
                        end
                    end
                end
                
                if not foundTarget and #checkpoints > 0 then
                    local nextPoint = checkpoints[currentCpIndex]
                    hrp.CFrame = CFrame.new(nextPoint)
                    
                    currentCpIndex = currentCpIndex + 1
                    if currentCpIndex > #checkpoints then
                        currentCpIndex = 1
                    end
                    
                    task.wait(TELE_DELAY_CP) 
                elseif not foundTarget and #checkpoints == 0 then
                    task.wait(0.5)
                end
            end
        end
    end
end)
